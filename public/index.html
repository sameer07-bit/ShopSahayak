 How to integrate into your existing project

            
You can call the Gemini OCR engine in 2 simple ways:

            
            

              

                Option 1: Import core service module directly (Node.js / Express / Next.js)
                
              

              
const { recognizeHandwriting } = require('./services/geminiOcrService');
const fs = require('fs');

async function processHandwrittenDocument() {
  const imageBuffer = fs.readFileSync('./path/to/handwritten_note.jpg');
  
  const result = await recognizeHandwriting({
    imageBuffer,
    mimeType: 'image/jpeg',
    apiKey: process.env.GEMINI_API_KEY,
    targetLanguage: 'English' // Optional: auto translate
  });

  console.log('Recognized Text:', result.fullText);
  console.log('Detected Language:', result.detectedLanguage);
  console.log('Line Breakdown:', result.lines);
}

processHandwrittenDocument();

            


            

              

                Option 2: REST API Endpoint (React, Vue, Mobile Apps, Python, cURL)
                
              

              
// Client-side upload using FormData
const formData = new FormData();
formData.append('image', fileInput.files[0]);
formData.append('targetLanguage', 'English');

const response = await fetch('/api/ocr/recognize', {
  method: 'POST',
  body: formData
});

const data = await response.json();
console.log('Recognized Text:', data.fullText);
